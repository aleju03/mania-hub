# broader Disallow (the /api/og exception below). A dedicated group means
# Twitterbot never reads the * group at all: it only fetches URLs humans
# share, so there is nothing to disallow for it. Discord/Facebook ignore
# robots.txt entirely and were never affected.

Allow: /

User-agent: *
Allow: /api/og
Disallow: /api/
